/**
 * useAIInterviewer Hook
 * 
 * React hook for managing the AI interviewer conversation flow
 * Provides easy-to-use interface for components to interact with the AI interviewer
 */

import { useState, useCallback, useRef, useEffect } from 'react';
import { createAIInterviewer } from '../services/aiInterviewer';
import speechService from '../services/speechService';
import speechRecognitionService from '../services/speechRecognitionService';
import audioRecorderService from '../services/audioRecorderService';
import { transcribeWithFallback, checkLocalWhisperHealth } from '../services/localWhisperService';
import InterviewBackendSync from '../services/interviewBackendSync';

export const useAIInterviewer = (config = {}) => {
  const [isInitialized, setIsInitialized] = useState(false);
  const [currentMessage, setCurrentMessage] = useState('');
  const [phase, setPhase] = useState('not_started'); // not_started, introduction, questions, candidate_questions, closing, completed
  const [isProcessing, setIsProcessing] = useState(false);
  const [isSpeaking, setIsSpeaking] = useState(false);
  const [isListening, setIsListening] = useState(false); // active capture (browser STT or recording)
  const [currentTranscript, setCurrentTranscript] = useState(''); // interim transcript (browser STT)
  const [whisperAvailable, setWhisperAvailable] = useState(false); // local whisper health
  const [isTranscribing, setIsTranscribing] = useState(false); // uploading + waiting whisper
  const [error, setError] = useState(null);
  const [conversationHistory, setConversationHistory] = useState([]);
  const [interviewState, setInterviewState] = useState(null);
  const [canCandidateSpeak, setCanCandidateSpeak] = useState(false);
  
  const interviewerRef = useRef(null);
  const backendSyncRef = useRef(null);
  const currentQuestionIdRef = useRef(null); // Track current question ID from backend

  useEffect(() => {
    return () => {
      backendSyncRef.current?.destroy?.();
      speechRecognitionService.stop();
      audioRecorderService.abort?.();
      speechService.cancel?.();
    };
  }, []);

  /**
   * Speak a message using TTS
   */
  const speakMessage = useCallback(async (message) => {
    if (!message) return;
    
    try {
      setIsSpeaking(true);
      setCanCandidateSpeak(false); // AI is speaking, candidate cannot
      
      await speechService.speak(message, {
        rate: 0.95,
        pitch: 1.0,
        volume: 1.0,
        onEnd: () => {
          setIsSpeaking(false);
          setCanCandidateSpeak(true); // AI finished, candidate can now speak
        },
        onError: (error) => {
          console.error('TTS error:', error);
          setIsSpeaking(false);
          setCanCandidateSpeak(true); // On error, allow candidate to speak
        }
      });
    } catch (err) {
      console.error('Speech error:', err);
      setIsSpeaking(false);
      setCanCandidateSpeak(true);
    }
  }, []);

  /**
   * Start listening to candidate's speech
   */
  const startListening = useCallback(async () => {
    if (!canCandidateSpeak || isListening) return false;

    // If Whisper is available -> record raw audio, else fallback to browser STT
    if (whisperAvailable) {
      const started = await audioRecorderService.start();
      if (!started) return false;
      setIsListening(true);
      setCurrentTranscript('');
      return true;
    }

    // Browser STT fallback
    speechRecognitionService.onResult = (result) => {
      setCurrentTranscript(result.full);
    };
    speechRecognitionService.onSilenceDetected = () => {
      stopListening();
    };
    speechRecognitionService.onStart = () => {
      setIsListening(true);
      setCurrentTranscript('');
    };
    speechRecognitionService.onEnd = () => setIsListening(false);
    speechRecognitionService.onError = (err) => {
      console.error('Speech recognition error:', err);
      setIsListening(false);
    };
    const started = speechRecognitionService.start();
    if (started) setIsListening(true);
    return started;
  }, [canCandidateSpeak, isListening, whisperAvailable]);

  /**
   * Stop listening to candidate's speech
   */
  const stopListening = useCallback(async () => {
    if (!isListening) return null;

    // Whisper path
    if (whisperAvailable && audioRecorderService.isRecording) {
      const blob = await audioRecorderService.stop();
      setIsListening(false);
      setIsTranscribing(true);
      try {
        const result = await transcribeWithFallback(blob, { language: 'en' });
        setIsTranscribing(false);
        return result.text;
      } catch (err) {
        console.warn('Whisper transcription failed, falling back:', err.message);
        setIsTranscribing(false);
        // Fallback: no transcript -> user retry
        return '';
      }
    }

    // Browser STT path
    speechRecognitionService.stop();
    const transcript = speechRecognitionService.getTranscript();
    setIsListening(false);
    const final = transcript.final;
    setCurrentTranscript('');
    return final;
  }, [isListening, whisperAvailable]);

  /**
   * Initialize the AI interviewer
   */
  const initializeInterview = useCallback(async (interviewConfig = {}) => {
    try {
      setIsProcessing(true);
      setError(null);

      // Check local whisper server once at init
      try {
        const healthy = await checkLocalWhisperHealth();
        setWhisperAvailable(healthy);
      } catch {
        setWhisperAvailable(false);
      }

      // Initialize backend sync if interviewId is provided
      const interviewId = config.interviewId || interviewConfig.interviewId;
      if (interviewId) {
        try {
          backendSyncRef.current = new InterviewBackendSync(interviewId);
          await backendSyncRef.current.initialize();
          
          // Start interview on backend if not already started
          await backendSyncRef.current.startInterview();
          
          // Get questions from backend
          const backendQuestions = backendSyncRef.current.getQuestions();
          if (backendQuestions && backendQuestions.length > 0) {
            // Update config with backend question count
            interviewConfig.totalQuestions = backendQuestions.length;
          }

          backendSyncRef.current.subscribeToRealtime((event) => {
            if (event?.eventType === 'interview-ended') {
              setPhase('completed');
              setCanCandidateSpeak(false);
            }
          });
        } catch (syncError) {
          console.warn('Backend sync failed, continuing with local interview:', syncError);
          // Continue without backend sync
        }
      }

      // Create interviewer instance
      const interviewer = createAIInterviewer({
        ...config,
        ...interviewConfig
      });
      
      interviewerRef.current = interviewer;

      // Start the interview
      const response = await interviewer.startInterview();
      
      setCurrentMessage(response.message);
      setPhase(response.phase);
      setIsInitialized(true);
      setInterviewState(interviewer.getState());
      
      // Add to conversation history
      setConversationHistory([{
        role: 'interviewer',
        message: response.message,
        timestamp: new Date().toISOString()
      }]);

      // Speak the welcome message
      await speakMessage(response.message);

      return response;
    } catch (err) {
      console.error('Failed to initialize interview:', err);
      setError(err.message);
      throw err;
    } finally {
      setIsProcessing(false);
    }
  }, [config]);

  /**
   * Send candidate's introduction
   */
  const sendIntroduction = useCallback(async (introduction) => {
    if (!interviewerRef.current) {
      throw new Error('Interview not initialized');
    }

    try {
      setIsProcessing(true);
      setError(null);

      // Add candidate message to history
      setConversationHistory(prev => [...prev, {
        role: 'candidate',
        message: introduction,
        timestamp: new Date().toISOString()
      }]);

      const response = await interviewerRef.current.processIntroduction(introduction);
      
      // Get first question from backend and mark it as asked
      if (backendSyncRef.current) {
        const nextQuestion = backendSyncRef.current.getNextQuestion();
        if (nextQuestion) {
          currentQuestionIdRef.current = nextQuestion.question.id;
          // Mark question as asked
          await backendSyncRef.current.markQuestionAsked(nextQuestion.question.id);
        }
      }
      
      setCurrentMessage(response.message);
      setPhase(response.phase);
      setInterviewState(interviewerRef.current.getState());

      // Add interviewer response to history
      setConversationHistory(prev => [...prev, {
        role: 'interviewer',
        message: response.message,
        timestamp: new Date().toISOString(),
        questionNumber: response.questionNumber
      }]);

      // Speak the question
      await speakMessage(response.message);

      return response;
    } catch (err) {
      console.error('Failed to process introduction:', err);
      setError(err.message);
      throw err;
    } finally {
      setIsProcessing(false);
    }
  }, []);

  /**
   * Send answer to interview question
   */
  const sendAnswer = useCallback(async (answer, audioUrl = null) => {
    if (!interviewerRef.current) {
      throw new Error('Interview not initialized');
    }

    try {
      setIsProcessing(true);
      setError(null);

      // Add candidate message to history
      setConversationHistory(prev => [...prev, {
        role: 'candidate',
        message: answer,
        timestamp: new Date().toISOString()
      }]);

      // Process answer with AI interviewer
      const response = await interviewerRef.current.processAnswer(answer);
      
      // Save answer to backend if backend sync is available
      if (backendSyncRef.current && currentQuestionIdRef.current) {
        try {
          const backendResponse = await backendSyncRef.current.submitAnswer(
            currentQuestionIdRef.current,
            answer,
            audioUrl
          );
          
          // Merge backend evaluation with AI response if available
          if (backendResponse.evaluation) {
            response.evaluation = {
              ...response.evaluation,
              ...backendResponse.evaluation,
              backendScore: backendResponse.evaluation.score,
            };
          }
        } catch (backendError) {
          console.warn('Failed to save answer to backend:', backendError);
          // Continue even if backend save fails
        }
      }
      
      setCurrentMessage(response.message);
      setPhase(response.phase);
      setInterviewState(interviewerRef.current.getState());

      // Add interviewer response to history
      setConversationHistory(prev => [...prev, {
        role: 'interviewer',
        message: response.message,
        timestamp: new Date().toISOString(),
        questionNumber: response.questionNumber,
        actionType: response.actionType,
        evaluation: response.evaluation
      }]);

      // Get next question ID from backend if available
      if (backendSyncRef.current && response.actionType === 'next_question') {
        const nextQuestion = backendSyncRef.current.getNextQuestion();
        if (nextQuestion) {
          currentQuestionIdRef.current = nextQuestion.question.id;
          // Mark question as asked
          await backendSyncRef.current.markQuestionAsked(nextQuestion.question.id);
        }
      }

      // Speak the response (question/follow-up/feedback)
      await speakMessage(response.message);

      return response;
    } catch (err) {
      console.error('Failed to process answer:', err);
      setError(err.message);
      throw err;
    } finally {
      setIsProcessing(false);
    }
  }, []);

  /**
   * Ask a question to the interviewer (when candidate asks about role/company)
   */
  const askQuestion = useCallback(async (question) => {
    if (!interviewerRef.current) {
      throw new Error('Interview not initialized');
    }

    try {
      setIsProcessing(true);
      setError(null);

      // Add candidate question to history
      setConversationHistory(prev => [...prev, {
        role: 'candidate',
        message: question,
        timestamp: new Date().toISOString(),
        type: 'candidate_question'
      }]);

      const response = await interviewerRef.current.answerCandidateQuestion(question);
      
      setCurrentMessage(response.message);
      setPhase(response.phase);
      setInterviewState(interviewerRef.current.getState());

      // Add interviewer response to history
      setConversationHistory(prev => [...prev, {
        role: 'interviewer',
        message: response.message,
        timestamp: new Date().toISOString(),
        type: 'answer_to_candidate'
      }]);

      // Speak the answer
      await speakMessage(response.message);

      return response;
    } catch (err) {
      console.error('Failed to ask question:', err);
      setError(err.message);
      throw err;
    } finally {
      setIsProcessing(false);
    }
  }, []);

  /**
   * End the interview
   */
  const endInterview = useCallback(async () => {
    if (!interviewerRef.current) {
      throw new Error('Interview not initialized');
    }

    try {
      setIsProcessing(true);
      setError(null);

      const response = await interviewerRef.current.concludeInterview();

      let backendInterview = null;
      if (backendSyncRef.current) {
        try {
          const backendResponse = await backendSyncRef.current.endInterview();
          backendInterview = backendResponse?.interview || null;

          if (backendInterview?.evaluation) {
            response.backendEvaluation = backendInterview.evaluation;
          }
          if (typeof backendInterview?.overallScore === 'number') {
            response.backendOverallScore = backendInterview.overallScore;
          }
          if (backendInterview?.readinessLevel) {
            response.backendReadinessLevel = backendInterview.readinessLevel;
          }
        } catch (backendError) {
          console.warn('Failed to end interview on backend:', backendError);
        }
      }
      
      setCurrentMessage(response.message);
      setPhase('completed');
      setInterviewState(interviewerRef.current.getState());

      // Add closing message to history
      setConversationHistory(prev => [...prev, {
        role: 'interviewer',
        message: response.message,
        timestamp: new Date().toISOString(),
        type: 'closing'
      }]);

      // Speak the closing message
      await speakMessage(response.message);

      return {
        ...response,
        backendInterview,
      };
    } catch (err) {
      console.error('Failed to end interview:', err);
      setError(err.message);
      throw err;
    } finally {
      setIsProcessing(false);
    }
  }, []);

  /**
   * Get full interview data for export
   */
  const exportInterview = useCallback(() => {
    if (!interviewerRef.current) {
      return null;
    }
    return interviewerRef.current.exportConversation();
  }, []);

  /**
   * Reset interview
   */
  const resetInterview = useCallback(() => {
    backendSyncRef.current?.destroy?.();
    interviewerRef.current = null;
    backendSyncRef.current = null;
    currentQuestionIdRef.current = null;
    setIsInitialized(false);
    setCurrentMessage('');
    setPhase('not_started');
    setConversationHistory([]);
    setInterviewState(null);
    setError(null);
  }, []);

  /**
   * Clear error
   */
  const clearError = useCallback(() => {
    setError(null);
  }, []);

  return {
    // State
    isInitialized,
    currentMessage,
    phase,
    isProcessing,
    isSpeaking,
    isListening,
    currentTranscript,
    whisperAvailable,
    isTranscribing,
    canCandidateSpeak,
    error,
    conversationHistory,
    interviewState,
    
    // Actions
